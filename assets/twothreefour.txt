void split(int idx) {
    ensureParent(idx);
    int upVal = nodes[idx].keys[2];
    int pIdx = nodes[idx].pIdx;
    int sibIdx = createSibling(idx);
    linkToParent(pIdx, upVal, sibIdx);
    if (nodes[pIdx].keys.size() > 3) {
        split(pIdx);
    }
}

void add(int nodeIdx, int val) {
    Node& nod = nods[nodeIdx];
    if (nod.children.empty()) {
        nod.keys.push_back(val);
        std::sort(nod.keys.begin(), nod.keys.end());
        std::erase(
            std::unique(nod.keys.begin(), nod.keys.end()), 
            nod.keys.end()
        );
        if (nod.keys.size() > 3) {
            split(nodeIdx);
        }
    } else {
        int i = 0;
        while (i < nod.keys.size() && val > nod.keys[i]) {
            i++;
        }
        add(nod.children[i], val);
    }
}

void del(int nodeIdx, int val) {
    Node& nod = nods[nodeIdx];
    for (auto& t: nod.keys) {
        if (t == val) {
            nod.keys.erase(t);
            return;
        }
    }
}

int find(int nodeIdx, int val) {
    Node& nod = nods[nodeIdx];
    for (int k: nod.keys) {
        if (k == val) {
            return nodeIdx;
        }
    }
    if (nod.children.empty()) {
        return -1;
    }
    int i = 0;
    while (i < nod.keys.size() && val > nod.keys[i]){
        i++;
    }
    return find(nod.children[i], val);
}
