struct LinearProbing {
    int n;
    vector<int> vals;
}

void init(int size) {
    n = size; 
    vals.assign(n, -1); 
}

int hash(int s) {
    int val = 0;
    while (s > 0) {
        val += (s % 10 + 1) * 293;
        val %= n;
    }
    return val;
}

int add(int s) {
    int i = hash(s);
    while (!vals[i].empty()) {
        if (vals[i][0] == -1) {
            break;
        }
        i = (i + 1) % n;
    }
    return vals[i] = s, i;
}

int del(int s) {
    int i = hash(s);
    for (int j = 0; j < n; j++) {
        if (vals[i] == s) {
            return vals[i] = -1, i;
        }
        if (vals[i].empty()) {
            break;
        }
        i = (i + 1) % n;
    }
    return -1;
}

int search(const string& s) {
    int i = hash(s);
    for (int j = 0; j < n; j++) {
        if (vals[i] == s) {
            return i;
        }
        if (vals[i].empty()) {
            break;
        }
        i = (i + 1) % n;
    }
    return -1;
}
