struct Node {
    int x[128], c = 0;
    Node() {
        for (int i = 0; i < 128; ++i) {
            x[i] = -1;
        }
    }
};

struct Trie {
    vector<Node> t;
    Trie() { 
        t.emplace_back(); 
    }
}

int add(const string& s) {
    int p = 0;
    for (char c: s) {
        int& n = t[p].x[(int)c];
        if (n == -1) {
            n = t.size();
            t.emplace_back();
        }
        p = n;
        t[p].c++;
    }
    return p;
}

void del(const string& s) {
    if (find(s) == -1) return;
    int p = 0;
    for (char c : s) {
        p = t[p].x[(int)c];
        t[p].c--;
    }
}

int find(const string& s) {
    int p = 0;
    for (char c : s) {
        p = t[p].x[(int)c];
        if (p < 0 || !t[p].c) {
            return -1;
        }
    }
    return p;
}
